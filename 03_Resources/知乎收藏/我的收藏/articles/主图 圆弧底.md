AA:=(C\*2+H+L+O)/5;

V1:=MA(C,5);

V2:(EMA(AA,4)+MA(AA,8)+MA(AA,16))/3,COLORWHITE,LINETHICK2;

V5:(EMA(AA,9)+MA(AA,18)+MA(AA,36))/3,COLORFF00FF,LINETHICK2;

V4:(EMA(AA,13)+MA(AA,26)+MA(AA,52))/3,COLOR00FF00,LINETHICK2;

V6:(EMA(AA,24)+MA(AA,48)+MA(AA,96))/3,COLOR0000FF,LINETHICK2;

V7:=EMA(100\*(C-LLV(L,100))/(HHV(H,100)-LLV(L,100)),5);

V8:=V7/4;

V10:=MAX(MAX(V2,V5),MAX(V4,V6));

V9:=MIN(MIN(V2,V5),MIN(V4,V6));

V16:=MIN(MIN(V2,V5),V4);

V20:=MAX(MAX(V2,V5),V4);

黄金叉:=CROSS(V5,V6)AND V4>REF(V4,1)AND V1>REF(V1,2)AND (V2-V6)/V6<0.02;

穿四线:=FILTER(H>V10 AND L<V9 AND V2>REF(V2,1)AND V5>REF(V5,1) AND V6>REF(V6,2)AND (V10-V9)/V9<0.04 AND C>O AND (H-C)/(C-O)<0.4,3);

圆弧底:=LONGCROSS(V2,V4,30)AND (V6-V4)/V4<0.05 AND V20<V6;

双交叉:=LONGCROSS(V2,V5,15) AND CROSS(V5,V4);

三线粘合:=FILTER(V6>REF(V6,2)AND REF(V2,3)< REF(V4,3)AND C>O AND(V20-V16)/V16<0.005,5);

DRAWICON(黄金叉 OR 双交叉 OR 穿四线 OR 三线粘合,L-0.1,9);

DRAWICON(圆弧底,L-0.1,5);
